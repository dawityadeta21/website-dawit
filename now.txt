        }

        function handleHeaderSearch(e) {
            const query = (e.target.value || '').toLowerCase().trim();
            if (query.length < 2) {
                const searchTab = document.getElementById('tab-search-portal');
                if (searchTab && searchTab.classList.contains('active')) {
                    document.getElementById('global-search-results').innerHTML = `<p style="grid-column: 1 / -1; text-align:center; color:#64748b; padding:20px;">Type at least 2 characters to search across modules, textbooks, and files.</p>`;
                }
                return;
            }

            switchCourseTab(null, 'search-portal');
            const resultsContainer = document.getElementById('global-search-results');
            if (!resultsContainer) return;

            let matches = [];

            // 1. Search Curriculum Modules & Courses
            Object.entries(disciplineData).forEach(([discKey, discObj]) => {
                discObj.courses.forEach((c, idx) => {
                    if (c.label.toLowerCase().includes(query) || c.summary.toLowerCase().includes(query)) {
                        matches.push({
                            title: `[${discKey.toUpperCase()}] ${c.label}`,
                            desc: c.summary,
                            meta: `Exam Weight: ${c.weight}`,
                            actionText: '📖 Study Module',
                            actionFunc: () => {
                                changeDiscipline(discKey);
                                switchCourseTab(null, c.id);
                            }
                        });
                    }
                });
            });

            // 2. Search Textbooks
            ehpleTextbooks.forEach(b => {
                if (b.title.toLowerCase().includes(query) || b.desc.toLowerCase().includes(query) || b.discipline.toLowerCase().includes(query)) {
                    const lic = getDisciplineLicense(b.discipline);
                    matches.push({
                        title: `[Textbook - ${b.discipline.toUpperCase()}] ${b.title}`,
                        desc: b.desc,
                        meta: `Author: ${b.author} | Size: ${b.size}`,
                        actionText: lic.active ? '📖 Read Book Online' : '🔒 Locked (Unlock ' + b.discipline.toUpperCase() + ')',
                        actionFunc: () => {
                            if (lic.active) renderEmbeddedPdfModal(b.file, b.title);
                            else openPaywall(b.discipline);
                        }
                    });
                }
            });

            // 3. Search Server Updated Files
            serverUpdatedFiles.forEach(f => {
                const disc = detectDisciplineFromFile(f.name);
                const nameWithoutExt = f.name.replace(/\.[^/.]+$/, "");
                if (nameWithoutExt.toLowerCase().includes(query) || f.name.toLowerCase().includes(query)) {
                    matches.push({
                        title: `[Updated File - ${disc.toUpperCase()}] ${nameWithoutExt}`,
                        desc: `Server update file: ${f.name}`,
                        meta: `Size: ${f.size}`,
                        actionText: '📖 Read Online',
                        actionFunc: () => renderEmbeddedPdfModal('documents/updates/' + f.name, nameWithoutExt)
                    });
                }
            });

            if (matches.length === 0) {
                resultsContainer.innerHTML = `<p style="grid-column: 1 / -1; text-align:center; color:#64748b; padding:20px;">No results found matching "${query}".</p>`;
                return;
            }

            resultsContainer.innerHTML = '';
            matches.forEach((m, mIdx) => {
                const card = document.createElement('div');
                card.className = 'pdf-card';
                card.innerHTML = `
                    <div>
                        <div class="pdf-card-title">${m.title}</div>
                        <div class="pdf-card-desc">${m.desc}</div>
                        <div class="pdf-card-meta"><span>${m.meta}</span></div>
                    </div>
                    <div class="pdf-btn-group">
                        <button class="btn-download-hub" id="search-act-${mIdx}">${m.actionText}</button>
                    </div>
                `;
                resultsContainer.appendChild(card);
                card.querySelector(`#search-act-${mIdx}`).onclick = m.actionFunc;
            });
        }

        function renderEmbeddedPdfModal(pdfUrl, moduleTitle) {
            if (!activeUser) {
                alert("Account creation is mandatory to view documents.");
                switchToAccountTab();
                return;
            }
            let targetDisc = currentDiscipline;
            if (pdfUrl.includes('documents/books/')) {
                const foundBook = ehpleTextbooks.find(b => b.file === pdfUrl);
                if (foundBook) targetDisc = foundBook.discipline;
            } else if (pdfUrl.includes('documents/updates/')) {
                targetDisc = detectDisciplineFromFile(pdfUrl);
            } else if (pdfUrl.includes('midwifery')) {
                targetDisc = 'midwifery';
            } else if (pdfUrl.includes('nursing')) {
                targetDisc = 'nursing';
            } else if (pdfUrl.includes('pharmacy')) {
                targetDisc = 'pharmacy';
            } else if (pdfUrl.includes('lab')) {
                targetDisc = 'laboratory';
            }
            const lic = getDisciplineLicense(targetDisc);
            if (!lic.active && !pdfUrl.includes('documents/updates/')) {
                alert(`Document reading is locked for ${targetDisc.toUpperCase()}.`);
                openPaywall(targetDisc);
                return;
            }
            const modalHTML = `
                <div class="pdf-viewer-overlay" style="position:fixed; top:0; left:0; width:100%; height:100%; background:rgba(0,0,0,0.85); z-index:3000; display:flex; flex-direction:column; padding:20px;">
                    <div style="display:flex; justify-content:space-between; align-items:center; background:#1e293b; color:white; padding:12px 20px; border-radius:8px 8px 0 0;">
                        <h3 style="margin:0; font-size:1rem;">📖 Reader: ${moduleTitle}</h3>
                        <button onclick="closePdfModal()" style="background:#ef4444; color:white; border:none; padding:6px 12px; border-radius:4px; font-weight:bold; cursor:pointer;">Close ✕</button>
                    </div>
                    <div style="flex:1; background:white; border-radius:0 0 8px 8px; overflow:hidden;">
                        <object data="${pdfUrl}#toolbar=0&navpanes=0&scrollbar=0" type="application/pdf" width="100%" height="100%">
                            <iframe src="${pdfUrl}#toolbar=0&navpanes=0&scrollbar=0" width="100%" height="100%" style="border:none;"></iframe>
                        </object>
                    </div>
                </div>
            `;
            let modalWrapper = document.getElementById('pdf-modal-container');
            if (!modalWrapper) {
                modalWrapper = document.createElement('div');
                modalWrapper.id = 'pdf-modal-container';
                document.body.appendChild(modalWrapper);
            }
            modalWrapper.innerHTML = modalHTML;
        }

        function closePdfModal() {

            if (modalWrapper) modalWrapper.innerHTML = '';
        }

        function loadComments() {

            if (!container) return;

                if (formContainer) formContainer.style.display = 'none';
                if (loginPrompt) loginPrompt.style.display = 'block';
            }
            let commentsDb = [];
            try { commentsDb = JSON.parse(localStorage.getItem('ehple_comments_db') || '[]'); } catch(e) {}

                container.innerHTML = `<p style="text-align:center; color:#64748b; font-style:italic; padding:20px;">No comments yet.</p>`;

                        <div><strong style="color:#1e3a8a; font-size:0.92rem;">👤 ${c.authorName}</strong><span style="font-size:0.78rem; background:#eff6ff; color:#2563eb; padding:2px 8px; border-radius:12px; margin-left:8px; font-weight:600;">🏛️ ${c.institution}</span></div>

                contact = rawPhone.startsWith('+251') ? rawPhone : (rawPhone.startsWith('0') ? rawPhone : '0' + rawPhone);
            } else {
                contact = document.getElementById('reg-email').value.trim().toLowerCase();
                if (!contact || !contact.includes('@')) { alert("Enter a valid email."); return; }

                toggleAuthMode('login');
                return;
            }

        function loadQuestion(course) {
            if (!activeUser) return;
            const state = courseState[course];
            if (!state) return;
            const idx = state.currentIndex;
            const lic = getDisciplineLicense(currentDiscipline);
            if (idx > 3 && !lic.active) { openPaywall(currentDiscipline); return; }
            const q = state.questions[idx];
            document.getElementById(`counter-${course}`).innerText = `QUESTION ${idx + 1} OF 200`;
            document.getElementById(`qtext-${course}`).innerText = q.q;
            const optContainer = document.getElementById(`qopt-${course}`);
            optContainer.innerHTML = '';
            q.options.forEach((opt, oIdx) => {
                optContainer.innerHTML += `<label><input type="radio" name="ans_${course}" value="${oIdx}"> ${opt}</label>`;
            });
            document.getElementById(`fb-${course}`).style.display = 'none';
            document.getElementById(`ex-${course}`).style.display = 'none';
            document.getElementById(`next-btn-${course}`).style.display = 'none';
        }

        function submitAnswer(course) {
            if (!activeUser) { switchToAccountTab(); return; }
            const state = courseState[course];
            const lic = getDisciplineLicense(currentDiscipline);
            if (state.currentIndex > 3 && !lic.active) { openPaywall(currentDiscipline); return; }
            const selected = document.querySelector(`input[name="ans_${course}"]:checked`);
            const fb = document.getElementById(`fb-${course}`);
            const ex = document.getElementById(`ex-${course}`);
            if (!selected) { alert('Please select an option.'); return; }
            const userAns = parseInt(selected.value);
            const q = state.questions[state.currentIndex];
            fb.style.display = 'block';
            ex.style.display = 'block';
            ex.innerHTML = `<strong>Rationale:</strong> ${q.exp}`;
            if (userAns === q.answer) {
                fb.className = 'feedback correct';
                fb.innerHTML = 'Correct!';
            } else {
                fb.className = 'feedback incorrect';
                fb.innerHTML = `Incorrect. Correct option is ${String.fromCharCode(65 + q.answer)}.`;
            }
            document.getElementById(`next-btn-${course}`).style.display = 'inline-block';
        }

        function nextQuestion(course) {
